
const film=document.getElementById('film'),scrub=document.getElementById('scrub'),time=document.getElementById('time'),status=document.getElementById('status'),play=document.getElementById('play'),audio=document.getElementById('audio');
const duration=8;let attempt=0,downloadUrl;
function update(){scrub.value=film.currentTime;time.textContent=film.currentTime.toFixed(2)+' / '+duration.toFixed(2)+'s';document.querySelectorAll('.chapter').forEach((button,index,list)=>button.setAttribute('aria-current',String(film.currentTime>=Number(button.dataset.time)&&(!list[index+1]||film.currentTime<Number(list[index+1].dataset.time))))}
function pause(){attempt++;film.pause();play.textContent=film.muted?'Play muted':'Play with sound';status.textContent='Paused.'}
async function start(){const current=++attempt;if(film.currentTime>=duration-.02)film.currentTime=0;try{await film.play();if(current!==attempt)return;play.textContent='Pause';status.textContent=film.muted?'Playing muted.':'Playing with music and sound.'}catch{if(current!==attempt)return;pause();status.textContent='Playback did not start. Tap Restart.'}}
play.addEventListener('click',()=>film.paused?start():pause());document.getElementById('restart').addEventListener('click',()=>{pause();film.currentTime=0;update();start()});audio.addEventListener('click',()=>{film.muted=!film.muted;audio.textContent=film.muted?'Sound off':'Sound on';audio.setAttribute('aria-pressed',String(film.muted));if(film.paused)play.textContent=film.muted?'Play muted':'Play with sound';status.textContent=film.muted?'Sound muted.':'Sound enabled.'});scrub.addEventListener('input',()=>{pause();film.currentTime=Number(scrub.value);update()});document.querySelectorAll('.chapter').forEach(button=>button.addEventListener('click',()=>{pause();film.currentTime=Number(button.dataset.time);update();status.textContent='Scene ready. Tap Play.'}));film.addEventListener('ended',()=>{pause();scrub.value=duration;time.textContent=duration.toFixed(2)+' / '+duration.toFixed(2)+'s';status.textContent='Finished. Replay or download the pilot.'});film.addEventListener('seeked',update);film.addEventListener('error',()=>{pause();status.textContent='Video could not load. Download the MP4 to review.'});function tick(){if(!film.paused)update();requestAnimationFrame(tick)}tick();document.getElementById('download').addEventListener('click',()=>{if(!downloadUrl){const bytes=Uint8Array.from(atob(film.getAttribute('src').split(',')[1]),char=>char.charCodeAt(0));downloadUrl=URL.createObjectURL(new Blob([bytes],{type:'video/mp4'}));document.getElementById('download').href=downloadUrl}status.textContent='Downloading the private review MP4.'});
